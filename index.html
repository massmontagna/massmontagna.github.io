import numpy as np

def yaw_rate_gain(v, L=2.8, k_us=0.005):
    """Steady-state yaw rate per rad of steer."""
    return v / (L + k_us * v**2)

v = np.linspace(0, 50, 100)  # speed, m/s
for k in (0.003, 0.005, 0.008):
    gain = yaw_rate_gain(v, k_us=k)
